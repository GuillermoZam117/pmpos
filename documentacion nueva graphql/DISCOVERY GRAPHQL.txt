#### Proceso para CREAR UN NUEVO TICKET 
### 1. Registrar Terminal 
mutation {
  registerTerminal(
    ticketType: "COMEDOR",
    terminal: "SERVIDOR",
    department: "MESAS",
    user: "CAJERO"
  )
}

### 2. Crear Ticket Terminal 
mutation CreateTerminalTicket {
  createTerminalTicket(terminalId: "lvVWgQcGYUu6f3Lam3BmvA") {
    uid
    totalAmount
  }
}


### 3.  Añadir ordenes al ticket terminal 
#pollo
mutation AddPOLLO {
  addOrderToTerminalTicket(
    terminalId: "lvVWgQcGYUu6f3Lam3BmvA"
    productName: "POLLO"
    portion: "Normal"
    quantity: 1
  ) { totalAmount }
}

#papas
mutation AddPAPAS {
  addOrderToTerminalTicket(
    terminalId: "lvVWgQcGYUu6f3Lam3BmvA"
    productName: "PAPAS AL AJILLO"
    portion: "ORD"
    quantity: 1
  ) { totalAmount }
}

### 4. Asignar mesa a terminal ticket 
mutation {
  changeEntityOfTerminalTicket(
    terminalId: "lvVWgQcGYUu6f3Lam3BmvA",
    type: "Mesas",
    name: "12"
  ) {
    id
    entities { name type }
  }
}

### 5. verificacion de mesa añadida al ticket 
query {
  getTerminalTicket(terminalId: "lvVWgQcGYUu6f3Lam3BmvA") {
    id
    entities { name type }
  }
}


### 6.  Ver ordenes del ticket terminal 

query { 
		getTerminalTicket(terminalId: "lvVWgQcGYUu6f3Lam3BmvA")
		{ 
			id 
			orders 
				{ uid } } 
		}  
		
		
### Imprimir factura (OPCIONAL PARA EL FLUJO DE NUEVO TICKET) ( Ejecutar comando de uaotmatizacion con foco en ticket uid) 

mutation { 
          executeAutomationCommandForTerminalTicket(terminalId: "lvVWgQcGYUu6f3Lam3BmvA",
													name: "Imprimir factura",
													value: "") 
														{ id }
		}


### 7. Añadir comentarios a pedidos antes de cerrar ticket (OPCIONAL EL USUARIO PUEDE O NO PUEDE INGRESAR COMENTARIOS A UN PEDIDO )

mutation { executeAutomationCommandForTerminalTicket(
														terminalId: "W6yjzeD8pk28J49lh1MReA",
														name: "TagOrder",
														value: "Comentario desde GraphiQL",
														
														orderUid: "-DLIb_sKeUyNIPluRCYF5A")
															{ id }
															}
		
		
### 8. Cerrar ticket terminal (imprimir comando y enviar pedido a cocina)  (AQUI FINALIZA EL CICLO DE CREACION DE NUEVO TICKET, DONDE ENVIAMOS LOS PEDIDOS A COCINA, LO CERRAMOS Y SE CAMBIAN DE ESTADO TICKET, ORDENES Y ENTIDAD)
		mutation {
  closeTerminalTicket(terminalId: "lvVWgQcGYUu6f3Lam3BmvA")
}

### Quitar registro de terminal (al cerrar el servidor o el logueo del usuario) 

mutation {
  unregisterTerminal(terminalId: "lvVWgQcGYUu6f3Lam3BmvA")
}

### Cancelar pedido (antes de cerrar el ticket o enviarlo a cocina 

mutation {
            executeAutomationCommandForTerminalTicket(terminalId: "lvVWgQcGYUu6f3Lam3BmvA",
																   orderUid: "9OzgPtIgm0a-6pgWg7v8Fg",
																   name: "Predeterminado Cancelar los pedidos",
                                                                   value: ""
																   ) 
																	{ id } 
																   }
